window.onload = function() {
    processInternalMatrix(rawAlliedData, 'allied');
    processInternalMatrix(rawGermanData, 'german');
    console.log("Internal wargame engines fully generated!");

function processInternalMatrix(rawDataString, roleLabel) {
    // Split the text block line by line
    const lines = rawDataString.split("\n");
    
    lines.forEach(line => {
        line = line.trim();
        if (!line || line.startsWith("#")) return; // Skip empty rows or comment notes
        
        // Split our columns using the pipe symbol (|)
        const parts = line.split("|");
        if (parts.length < 3) return; // Guard against broken entries
        
        const pageNum = parseInt(parts[0].trim(), 10);
        const midTurnsArray = parts[1].trim().split(".").map(Number);
        
        // Initialize structural mapping slots
        if (!bookMatrices[roleLabel][pageNum]) {
            bookMatrices[roleLabel][pageNum] = {};
        }
        
        // Map all 26 maneuvers sequentially to their respective wizard keys
        moveKeyIndexMap.forEach((moveKey, index) => {
            bookMatrices[roleLabel][pageNum][moveKey] = {
                midTurn: midTurnsArray[index] || 170, // Fallback to 170 if cell blank
            };
        });
    });
}
