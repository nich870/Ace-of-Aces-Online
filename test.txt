const tutorialSlides = [
    {
        image: "images/tutorial_1.jpg",
        text: "<strong>Step 1: Synchronize</strong><br>Both players start on the exact same page number (typically Page 170). The image box displays what you see looking through your cockpit directly at the enemy plane."

    },

        text: "<strong>Step 3: Swap Mid-Turn Codes</strong><br>Once an option is tapped, read your generated Mid-Turn page number out loud to your opponent, and type their code into your input box. The devices cross-reference paths instantly!"
    },

        image: "images/tutorial_4.jpg",
        text: "<strong>Step 4: Resolve combat results</strong><br>Both screens jump in perfect sync to the next page. If a pilot lands in a combat sector cone, the app logs damage, updates stats, and fires structural hit alerts!"
    }
];

let currentTutorialIndex = 0;

// Universal Screen Swapping Manager Utility
function navigateTo(targetScreenId) {
    // Hide all structural panels safely
    const screens = document.querySelectorAll('.screen');
    screens.forEach(s => s.classList.remove('active'));
    
    // Unveil targeted dashboard screen element node
    document.getElementById(targetScreenId).classList.add('active');
}

// Initializing Flight School
function openTutorial() {
    currentTutorialIndex = 0;
    navigateTo('tutorial-screen');
    renderTutorialFrame();
}

function renderTutorialFrame() {
    const slide = tutorialSlides[currentTutorialIndex];
    
    // Update image and text blocks
    document.getElementById('tutorial-img').src = slide.image;
    document.getElementById('tutorial-text').innerHTML = slide.text;
    
    // Handle status header updates
    document.getElementById('tut-progress-indicator').innerText = `${currentTutorialIndex + 1} / ${tutorialSlides.length}`;
    
    // Tweak action button headers on the last frame to offer a clean exit
    const nextBtn = document.getElementById('tut-next-btn');
    if (currentTutorialIndex === tutorialSlides.length - 1) {
        nextBtn.innerText = "🏁 FINISH";
        nextBtn.style.background = "#28a745";
        nextBtn.style.borderColor = "#1e7e34";
    } else {
        nextBtn.innerText = "Next ➡";
        nextBtn.style.background = "#007bff";
        nextBtn.style.borderColor = "#0056b3";
    }
    
    // Toggle active display boundaries on back navigation controls
    document.getElementById('tut-prev-btn').style.visibility = (currentTutorialIndex === 0) ? 'hidden' : 'visible';
}

function changeTutorialPage(directionModifier) {
    currentTutorialIndex += directionModifier;
    
    // If the index pushes past the array limits, exit out back to the main menu screen
    if (currentTutorialIndex < 0 || currentTutorialIndex >= tutorialSlides.length) {
        navigateTo('main-menu-screen');
        return;
    }
    
    renderTutorialFrame();
}
